cmake_minimum_required(VERSION 3.20)
project(p3d_reader VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(P3D_BUILD_TOOLS "Build command-line inspection tool" ON)
option(P3D_BUILD_TESTS "Build regression tests" ON)
add_library(p3d_reader STATIC src/core.cpp src/records.cpp src/command_metadata.cpp src/attributes.cpp src/blobs.cpp src/scene.cpp src/geometry.cpp src/guided.cpp src/object_graph.cpp src/accessors.cpp src/data.cpp src/application.cpp src/electrical.cpp src/complex.cpp src/binary_formats.cpp src/definitions.cpp third_party/lz4/lz4.c third_party/pugixml/pugixml.cpp third_party/miniz/miniz_tinfl.c)
# Only tinfl is linked; unused zlib wrappers otherwise require miniz.c in Debug builds.
set_source_files_properties(third_party/miniz/miniz_tinfl.c PROPERTIES COMPILE_DEFINITIONS MINIZ_NO_ZLIB_APIS)
add_library(p3d::reader ALIAS p3d_reader)
set_target_properties(p3d_reader PROPERTIES EXPORT_NAME reader)
target_compile_features(p3d_reader PUBLIC cxx_std_17)
target_include_directories(p3d_reader PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/third_party> $<INSTALL_INTERFACE:include> PRIVATE src)
if(WIN32)
 target_compile_definitions(p3d_reader PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
if(MSVC)
 target_compile_options(p3d_reader PRIVATE /utf-8 /W4)
else()
 target_compile_options(p3d_reader PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wno-misleading-indentation>)
endif()
find_package(Threads REQUIRED)
target_link_libraries(p3d_reader PUBLIC Threads::Threads)
if(NOT WIN32)
 find_package(Iconv REQUIRED)
 target_link_libraries(p3d_reader PRIVATE Iconv::Iconv)
endif()
if(P3D_BUILD_TOOLS)
 add_executable(p3d-inspect tools/inspect.cpp)
 target_link_libraries(p3d-inspect PRIVATE p3d_reader)
 if(MINGW)
  target_link_options(p3d-inspect PRIVATE -static)
 endif()
 if(WIN32)
  target_link_libraries(p3d-inspect PRIVATE shell32 psapi)
 endif()
endif()
if(P3D_BUILD_TESTS)
 enable_testing()
 add_executable(p3d-unit tests/unit.cpp)
 target_include_directories(p3d-unit PRIVATE src)
 target_link_libraries(p3d-unit PRIVATE p3d_reader)
 add_test(NAME p3d-unit COMMAND p3d-unit)
endif()
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/p3dConfig.cmake.in
 ${CMAKE_CURRENT_BINARY_DIR}/p3dConfig.cmake
 INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/p3d)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/p3dConfigVersion.cmake
 VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(TARGETS p3d_reader EXPORT p3dTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES third_party/nlohmann/json.hpp DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/nlohmann)
install(EXPORT p3dTargets NAMESPACE p3d:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/p3d)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/p3dConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/p3dConfigVersion.cmake
 DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/p3d)
install(FILES LICENSE THIRD_PARTY.md DESTINATION ${CMAKE_INSTALL_DATADIR}/p3d)
install(DIRECTORY third_party/ DESTINATION ${CMAKE_INSTALL_DATADIR}/p3d/third_party
 FILES_MATCHING PATTERN "*LICENSE*" PATTERN "*COPYING*" PATTERN "manifest.json")
